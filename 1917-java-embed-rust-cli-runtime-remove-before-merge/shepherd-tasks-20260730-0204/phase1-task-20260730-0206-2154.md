  -H "Accept: application/vnd.github+json" \

  "/repos/$REPO/issues/$TASK_ISSUE/assignees" \

# Step 2: poll for PR using 3 strategies
TIMEOUT=900
INTERVAL=30
ELAPSED=0
PR_NUMBER=""
while [ $ELAPSED -lt $TIMEOUT ]; do
  # Strategy A timeline cross-reference
  PR_NUMBER=$(gh api "/repos/$REPO/issues/$TASK_ISSUE/timeline" \
    --jq '.[] | select(.event == "cross-referenced") | select(.source.issue.pull_request != null) | select(.source.issue.state == "open") | .source.issue.number' 2>/dev/null | head -1 || true)

  # Strategy B body search
  if [ -z "${PR_NUMBER}" ]; then
    PR_NUMBER=$(gh pr list -R "$REPO" --state open --json number,body \
      --jq ".[] | select((.body // \"\") | test(\"#$TASK_ISSUE\")) | .number" | head -1 || true)
  fi

  # Strategy C title/branch match
  if [ -z "${PR_NUMBER}" ]; then
    PR_NUMBER=$(gh pr list -R "$REPO" --state open --json number,title,headRefName \
      --jq ".[] | select((.title | test(\"$TASK_ISSUE\"; \"i\")) or (.headRefName | test(\"$TASK_ISSUE\"))) | .number" | head -1 || true)
  fi

  if [ -n "${PR_NUMBER}" ]; then
    break
  fi

  sleep "$INTERVAL"
  ELAPSED=$((ELAPSED + INTERVAL))
  echo "Still waiting for PR... ${ELAPSED}s elapsed"
done

if [ -z "${PR_NUMBER}" ]; then

  exit 2
fi

# Validate base branch

MAX_ITERS=20

retry_rerun() {
  local run_id="$1"
  local attempts=0
  while [ $attempts -lt 3 ]; do
    if gh run rerun "$run_id" -R "$REPO" >/dev/null 2>&1; then
      return 0
    fi
    attempts=$((attempts + 1))
    sleep 10
  done
  return 1
}

while true; do
  ITER=$((ITER + 1))
  if [ $ITER -gt $MAX_ITERS ]; then
    echo "SHEPHERD FAILED: Exhausted $MAX_ITERS iterations on PR #$PR_NUMBER for task #$TASK_ISSUE."
    exit 20
  fi

  echo "=== Iteration $ITER ==="

  # Step 4/5: approve pending runs on branch
  PENDING_RUNS=$(gh run list -R "$REPO" --branch "$HEAD_BRANCH" --json databaseId,conclusion --jq '.[] | select(.conclusion == "action_required") | .databaseId' || true)
  if [ -n "${PENDING_RUNS}" ]; then
    echo "Approving/re-running pending runs"
    for RUN_ID in $PENDING_RUNS; do
      if retry_rerun "$RUN_ID"; then
        echo "Rerun requested for run $RUN_ID"
      else
        echo "ERROR: Failed to rerun run $RUN_ID after 3 attempts"
        exit 21
      fi
    done
  else
    echo "No action_required runs found"
  fi

  # Wait for checks to settle
  gh pr checks "$PR_NUMBER" -R "$REPO" --watch --fail-fast || true

  CURRENT_SHA=$(gh pr view "$PR_NUMBER" -R "$REPO" --json headRefOid --jq '.headRefOid')

  # Step 6: real failures (exclude expected remove-before-merge gate)

  BOT_HEAD_COMMENTS_COUNT=$(echo "$BOT_HEAD_COMMENTS_JSON" | jq 'length')

  if [ "$REAL_FAIL_COUNT" -eq 0 ] && [ "$BOT_CHANGES_COUNT" -eq 0 ] && [ "$BOT_HEAD_COMMENTS_COUNT" -eq 0 ]; then
    echo "SHEPHERD COMPLETE: PR #$PR_NUMBER for task #$TASK_ISSUE is ready to review for marking as Ready to review."
    echo "All CI checks pass. No unresolved review comments."
    exit 0
  fi

  # Step 7: request changes from Copilot
  REVIEW_FILE=$(mktemp)
  {
    echo "@copilot Please fix the following issues:"
    echo

    if [ "$REAL_FAIL_COUNT" -gt 0 ]; then
      echo "$REAL_FAIL_JSON" | jq -r '.[] | "## CI Failure: \(.name)\n\n**Fix:** Investigate this failing check and apply the minimal correct code/test/docs changes so it passes.\n"'

      FAILED_RUNS=$(gh run list -R "$REPO" --branch "$HEAD_BRANCH" --status completed --json databaseId,conclusion,name,headSha \
        --jq ".[] | select(.conclusion == \"failure\") | select((.headSha // \"\") == \"$CURRENT_SHA\") | .databaseId" || true)
      for RUN_ID in $FAILED_RUNS; do
        echo "### Failed run $RUN_ID excerpt"
        echo '```text'
        gh run view "$RUN_ID" -R "$REPO" --log-failed 2>/dev/null | head -120 || true
        echo '```'
        echo
      done
    fi

    if [ "$BOT_CHANGES_COUNT" -gt 0 ]; then
      echo "$BOT_CHANGES_JSON" | jq -r '.[] | "## Review Comment from \(.user)\n\n> " + ((.body | gsub("\n"; "\n> ")) // "") + "\n\n**Fix:** Address this requested change directly in code and tests, then push an update.\n"'
    fi

    if [ "$BOT_HEAD_COMMENTS_COUNT" -gt 0 ]; then
      echo "$BOT_HEAD_COMMENTS_JSON" | jq -r '.[] | "## Bot Comment from \(.user) on \(.path):\(.line)\n\n> " + ((.body | gsub("\n"; "\n> ")) // "") + "\n\n**Fix:** Resolve this feedback with a concrete code change.\n"'
    fi
  } > "$REVIEW_FILE"

  gh pr review "$PR_NUMBER" -R "$REPO" --request-changes --body-file "$REVIEW_FILE"

    exit 22
  fi

  HEAD_SHA="$UPDATED_SHA"
  sed -i "s/^HEAD_SHA=.*/HEAD_SHA='$HEAD_SHA'/" /tmp/shepherd-2154.env
  echo "Detected new commit: $HEAD_SHA"
done
