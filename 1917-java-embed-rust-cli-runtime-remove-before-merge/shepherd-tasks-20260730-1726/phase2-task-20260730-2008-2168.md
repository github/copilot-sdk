161.     private static String readElfPtInterp(byte[] probe) throws IOException {

185.         } else if (elfClass == ELF_CLASS_32) {
186.             phoff = readUInt32(probe, 28, littleEndian);
187.             phentsize = readUInt16(probe, 42, littleEndian);
188.             phnum = readUInt16(probe, 44, littleEndian);

191.         }
192.

194.             throw new IOException("Program header table offset outside probe window: " + phoff);

197.             throw new IOException("Invalid ELF program header metadata: phentsize=" + phentsize + ", phnum=" + phnum);
198.         }
199. 
200.         for (int i = 0; i < phnum; i++) {

203.                 break;

205.             int base = (int) baseLong;
206.             if (base + phentsize > size) {
207.                 break;
208.             }
209.

219.                 pFileSize = readUInt64(probe, base + 32, littleEndian);
