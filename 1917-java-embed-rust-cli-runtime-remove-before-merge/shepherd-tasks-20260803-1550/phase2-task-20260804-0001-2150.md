30.  * directly through JNA.
31.  * </p>
32.  *
33.  * <p>

35.  * ({@code rust/tests/e2e/support.rs}) and the .NET

56. 
57.     /**

59.      * block.

64. 
65.     /** POSIX libc: sets or deletes a variable in the process environment block. */

71. 
72.     /**

77.     /**

88.             apply(entry.getKey(), entry.getValue());

95.         }

97. 
98.     private void apply(String name, String value) {
99.         String previous = System.getenv(name);
100.         saved.add(Map.entry(name, previous == null ? "" : previous));
101.         nativeSetEnv(name, value);
102.     }
103. 
104.     /**
105.      * Restores every environment variable this guard touched to the value it had
106.      * before construction.
107.      */
108.     @Override
109.     public void close() {

111.         Collections.reverse(reversed);
112.         for (Map.Entry<String, String> entry : reversed) {
113.             nativeSetEnv(entry.getKey(), entry.getValue().isEmpty() ? null : entry.getValue());
114.         }
115.     }
116. 
117.     private static void nativeSetEnv(String name, String value) {
118.         if (isWindows()) {

120.         } else {
121.             nativeSetEnvUnix(name, value);
122.         }
123.     }
124. 
125.     private static void nativeSetEnvWindows(String name, String value) {
126.         Kernel32Env kernel32 = Native.load("kernel32", Kernel32Env.class);
127.         boolean ok = kernel32.SetEnvironmentVariableW(new WString(name), value != null ? new WString(value) : null);
128.         if (!ok) {
129.             LOG.warning("SetEnvironmentVariableW failed for key=" + name);
130.         }
131.     }
132. 
133.     private static void nativeSetEnvUnix(String name, String value) {
134.         LibcEnv libc = Native.load("c", LibcEnv.class);
135.         if (value != null) {
136.             int rc = libc.setenv(name, value, 1);
137.             if (rc != 0) {
138.                 LOG.warning("setenv() failed for key=" + name + " rc=" + rc);

140.         } else {

142.             if (rc != 0) {
143.                 LOG.warning("unsetenv() failed for key=" + name + " rc=" + rc);
144.             }
145.         }
146.     }
147. 
148.     private static boolean isWindows() {
149.         return System.getProperty("os.name", "").toLowerCase(Locale.ROOT).contains("win");
150.     }
